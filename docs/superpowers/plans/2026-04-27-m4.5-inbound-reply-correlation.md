# M4.5 — Inbound Reply Correlation

**Goal:** Wire up reply detection so that when a prospect replies to a campaign email, the `replied` case already implemented in `emailEvent.service.ts` is actually triggered.

**Problem:** Neither Resend nor SendGrid fires a "reply" webhook from their outbound system. Reply detection requires *inbound email routing*: the workspace configures a reply-to address whose MX record points at Resend/SendGrid, which then delivers arriving emails as a POST to our server. We parse the `In-Reply-To` SMTP header from the inbound payload to correlate back to the sent message.

**Architecture:**

```
Prospect replies → MX → Resend/SendGrid inbound routing
    → POST /webhooks/inbound/resend (or /sendgrid)
    → processInboundEmail()
        → extractInReplyTo()     — parse In-Reply-To from inbound payload headers
        → normalizeMessageIdRef()  — strip <>, @domain, .filter suffix
        → SequenceEnrollment.findOne({ 'stepHistory.messageId': id })
        → handleReplyForEnrollment()  — refactored from existing replied case
        → fallback: OutreachDraft.findOne({ 'deliveryMetadata.messageId': id })
        → Campaign stat + notification for direct-send path
```

**Resend message ID correlation:**
- We store `data.id` (e.g. `"re_abc123"`) in `stepHistory[].messageId` at send time.
- Resend sets SMTP `Message-ID: <re_abc123@resend.dev>` on the actual email.
- When prospect replies, `In-Reply-To: <re_abc123@resend.dev>` appears in the inbound payload.
- Strip `<>` + `@resend.dev` → `"re_abc123"` → exact match in our DB.

**SendGrid message ID correlation:**
- We store `smtp-id.split('.')[0]` in `stepHistory[].messageId`.
- SMTP `Message-ID: <abc123.filterXXX@sg.sendgrid.net>` on sent email.
- Inbound parse: `In-Reply-To: <abc123.filterXXX@sg.sendgrid.net>` in headers string.
- Strip `<>` + `@domain` + `.split('.')[0]` → `"abc123"` → matches stored value.

```typescript
/**
 * Parse the local part of an RFC 5322 Message-ID reference.
 * "<re_abc@resend.dev>" → "re_abc"
 * "<abc123.filter001@sg.sendgrid.net>" → "abc123"
 */
function normalizeMessageIdRef(raw: string): string {
  const stripped = raw.trim().replace(/^<|>$/g, '');
  const atIdx = stripped.indexOf('@');
  const local = atIdx >= 0 ? stripped.slice(0, atIdx) : stripped;
  return local.split('.')[0] ?? local;
}

function extractInReplyTo(provider: 'resend' | 'sendgrid', payload: Record<string, unknown>): string | null {
  if (provider === 'resend') {
    const data = payload['data'] as Record<string, unknown> | undefined;
    const headers = (data?.['headers'] as Array<{ name: string; value: string }> | undefined) ?? [];
    const h = headers.find(h => h.name.toLowerCase() === 'in-reply-to');
    if (!h?.value) return null;
    const m = h.value.match(/<([^>]+)>/);
    return m ? `<${m[1]}>` : h.value.trim();
  }
  // SendGrid Inbound Parse: 'headers' is a CRLF-delimited text string
  const headersText = (payload['headers'] as string | undefined) ?? '';
  const m = headersText.match(/^In-Reply-To:\s*(<[^>]+>)/im);
  return m?.[1] ?? null;
}
```

- [ ] **Step 4: Add `processInboundEmail` export after `processEmailEvent`**

```typescript
/**
 * Handle an inbound email delivered via Resend or SendGrid inbound routing.
 * Parses In-Reply-To and correlates to a SequenceEnrollment (primary) or
 * OutreachDraft (fallback) to fire the `replied` campaign actions.
 */
export async function processInboundEmail(
  provider: 'resend' | 'sendgrid',
  payload: Record<string, unknown>,
): Promise<void> {
  const rawRef = extractInReplyTo(provider, payload);
  if (!rawRef) {
    logger.info('[emailEvent/inbound] no In-Reply-To header', { provider });
    return;
  }

  const messageId = normalizeMessageIdRef(rawRef);
  if (!messageId) {
    logger.warn('[emailEvent/inbound] could not normalize In-Reply-To', { rawRef });
    return;
  }

  const occurredAt = new Date();

  // Primary: sequence enrollment path
  const enrollment = await SequenceEnrollment.findOne({
    'stepHistory.messageId': messageId,
  });

  if (enrollment) {
    await EmailEvent.create({
      workspaceId: enrollment.workspaceId,
      enrollmentId: enrollment._id,
      messageId,
      event: 'replied',
      provider,
      raw: payload,
      occurredAt,
    });
    const stepIndex = enrollment.stepHistory.findIndex(s => s.messageId === messageId);
    await handleReplyForEnrollment(enrollment, messageId, occurredAt, stepIndex);
    return;
  }

  // Fallback: direct OutreachDraft send (no SequenceEnrollment)
  const draft = await OutreachDraft.findOne({
    'deliveryMetadata.messageId': messageId,
  }).select('_id campaignId leadId workspaceId');

  if (!draft) {
    logger.info('[emailEvent/inbound] no enrollment or draft for messageId', { messageId });
    await EmailEvent.create({ messageId, event: 'replied', provider, raw: payload, occurredAt });
    return;
  }

  await EmailEvent.create({
    workspaceId: draft.workspaceId,
    messageId,
    event: 'replied',
    provider,
    raw: payload,
    occurredAt,
  });

  const campaign = await Campaign.findOne({
    _id: draft.campaignId,
    workspaceId: draft.workspaceId,
  }).select('_id name');

  if (campaign) {
    await Campaign.updateOne({ _id: campaign._id }, { $inc: { 'stats.replied': 1 } });
    const lead = await Lead.findById(draft.leadId).select('companyName');
    await emitNotification({
      workspaceId: draft.workspaceId,
      type: 'campaign.reply',
      title: `Reply from ${lead?.companyName ?? 'a lead'}`,
      message: `${campaign.name ?? 'Campaign'} — reply received.`,
      href: `/dashboard/campaigns/${String(campaign._id)}`,
      metadata: {
        draftId: String(draft._id),
        leadId: String(draft.leadId),
        campaignId: String(campaign._id),
      },
    });
  }
}
```

```typescript
export async function handleResendInbound(req: Request, res: Response): Promise<void> {
  try {
    const payload = req.body as Record<string, unknown>;
    // Only process email.received events (Resend may also send email.bounced etc. on inbound)
    if (payload['type'] !== 'email.received') {
      res.status(200).json({ received: true });
      return;
    }
    await processInboundEmail('resend', payload);
  } catch (err) {
    logger.error('[webhooks] Resend inbound processing error', { err });
  }
  res.status(200).json({ received: true });
}

export async function handleSendGridInbound(req: Request, res: Response): Promise<void> {
  // SendGrid Inbound Parse sends multipart/form-data. Express must be configured
  // with a body parser that handles this (e.g. express-formidable or multer).
  // If req.body has 'headers', it was parsed; otherwise log and bail gracefully.
  try {
    const payload = req.body as Record<string, unknown>;
    if (!payload['headers']) {
      logger.warn('[webhooks] SendGrid inbound: missing headers field — check multipart body parser config');
      res.status(200).json({ received: true });
      return;
    }
    await processInboundEmail('sendgrid', payload);
  } catch (err) {
    logger.error('[webhooks] SendGrid inbound processing error', { err });
  }
  res.status(200).json({ received: true });
}
```
